AudioEngine::AudioEngine() {}
AudioEngine::~AudioEngine() { stop(); }

bool AudioEngine::start(const std::string& clientName)
{
    if (mRunning.load()) return true;

    jack_status_t status;
    mClient = jack_client_open(clientName.c_str(), JackNullOption, &status);
    if (!mClient)
    {
        std::cerr << "[AudioEngine] jack_client_open failed, status=0x"
                  << std::hex << status << "\n";
        return false;
    }

    mSampleRate = static_cast<double>(jack_get_sample_rate(mClient));
    const int maxBuf = static_cast<int>(jack_get_buffer_size(mClient));

    // Allocate work buffers
    mWorkBuf .resize(maxBuf, 0.0f);
    mWorkBufL.resize(maxBuf, 0.0f);
    mWorkBufR.resize(maxBuf, 0.0f);

    // Initialise all effects with sample rate
    mEQ     .setSampleRate(mSampleRate);
    mTuner  .setSampleRate(mSampleRate);
    mGate   .setSampleRate(mSampleRate);
    mComp   .setSampleRate(mSampleRate);
    mPitch  .setSampleRate(mSampleRate);
    mChorus1.setSampleRate(mSampleRate);
    mFlanger.setSampleRate(mSampleRate);
    mPhaser .setSampleRate(mSampleRate);
    mChorus2.setSampleRate(mSampleRate);
    mTremolo.setSampleRate(mSampleRate);
    mDelay  .setSampleRate(mSampleRate);
    mReverb .setSampleRate(mSampleRate);

    jack_set_process_callback(mClient, jackProcessCallback, this);
    jack_on_shutdown(mClient, jackShutdownCallback, this);

    mPortIn   = jack_port_register(mClient, "input",    JACK_DEFAULT_AUDIO_TYPE, JackPortIsInput,  0);
    mPortOutL = jack_port_register(mClient, "output_L", JACK_DEFAULT_AUDIO_TYPE, JackPortIsOutput, 0);
    mPortOutR = jack_port_register(mClient, "output_R", JACK_DEFAULT_AUDIO_TYPE, JackPortIsOutput, 0);

    if (!mPortIn || !mPortOutL || !mPortOutR)
    {
        std::cerr << "[AudioEngine] Failed to register JACK ports\n";
        jack_client_close(mClient);
        mClient = nullptr;
        return false;
    }

    if (jack_activate(mClient) != 0)
    {
        std::cerr << "[AudioEngine] jack_activate failed\n";
        jack_client_close(mClient);
        mClient = nullptr;
        return false;
    }

    mRunning.store(true);
    std::cout << "[AudioEngine] JACK started at " << mSampleRate << " Hz (stereo out)\n";
    return true;
}

void AudioEngine::stop()
{
    if (!mRunning.load()) return;
    mRunning.store(false);
    if (mClient)
    {
        jack_deactivate(mClient);
        jack_client_close(mClient);
        mClient = nullptr;
    }
}

// ---- Model management -------------------------------------------------

bool AudioEngine::loadModel(const std::string& path)
{
    std::unique_ptr<nam::DSP> newModel;
    try
    {
        newModel = nam::get_dsp(std::filesystem::path(path));
    }
    catch (const std::exception& e)
    {
        std::cerr << "[AudioEngine] Failed to load model: " << e.what() << "\n";
        return false;
    }
    if (!newModel) return false;

    newModel->Reset(mSampleRate, static_cast<int>(mWorkBuf.size()));
    newModel->prewarm();

    {
        std::lock_guard<std::mutex> lock(mModelMutex);
        int cur = mActiveModel.load();
        if (cur == 0) { mModelB = std::move(newModel); mActiveModel.store(1); mModelA.reset(); }
        else          { mModelA = std::move(newModel); mActiveModel.store(0); mModelB.reset(); }
        mModelPath = path;
        mModelLoaded.store(true);
    }
    return true;
}

void AudioEngine::unloadModel()
{
    std::lock_guard<std::mutex> lock(mModelMutex);
    mModelLoaded.store(false);
    mModelA.reset();
    mModelB.reset();
    mModelPath = "";
}

// ---- Dynamic effect order --------------------------------------------

void AudioEngine::setEffectOrder(const EffectOrder& order)
{
    std::lock_guard<std::mutex> lock(mOrderMutex);
    // Write to the inactive buffer, then flip
    int next = 1 - mOrderBuf.load(std::memory_order_relaxed);
    mOrders[next] = order;
    mOrderBuf.store(next, std::memory_order_release);
}

EffectOrder AudioEngine::currentOrder() const
{
    return mOrders[mOrderBuf.load(std::memory_order_acquire)];
}

// ---- Per-effect dispatch helpers -------------------------------------

void AudioEngine::processPreNam(int id, float* buf, int n)
{
    switch (id)
    {
        case EffectId::Tuner:
            if (mTunerEnabled.load())
                mTuner.processBlock(buf, n, mTunerMute.load());
            break;
        case EffectId::Gate:
            if (mGateEnabled.load())
                mGate.processBlock(buf, n);
            break;
        case EffectId::Comp:
            if (mCompEnabled.load())
                mComp.processBlock(buf, n);
            break;
        case EffectId::PitchShift:
            if (mPitchEnabled.load())
                mPitch.processBlock(buf, n);
            break;
        case EffectId::Chorus1:
            if (mChorus1Enabled.load())
                mChorus1.processBlock(buf, n);
            break;
        case EffectId::Flanger:
            if (mFlangerEnabled.load())
                mFlanger.processBlock(buf, n);
            break;
        case EffectId::Phaser:
            if (mPhaserEnabled.load())
                mPhaser.processBlock(buf, n);
            break;
        default: break;
    }
}

void AudioEngine::processPostNam(int id, float* L, float* R, int n)
{
    switch (id)
    {
        case EffectId::Chorus2:
            if (mChorus2Enabled.load())
                mChorus2.processBlockStereo(L, R, n);
            break;
        case EffectId::Tremolo:
            if (mTremoloEnabled.load())
                mTremolo.processBlockStereo(L, R, n);
            break;
        case EffectId::Delay:
            if (mDelayEnabled.load())
                mDelay.processBlockStereo(L, R, n);
            break;
        case EffectId::Reverb:
            if (mReverbEnabled.load())
                mReverb.processBlockStereo(L, R, n);
            break;
        case EffectId::IR:
            if (mIREnabled.load() && mIRLoader.isLoaded())
            {
                mIRLoader.processBlock(L, n);
                mIRLoader.processBlock(R, n);
            }
            break;
        default: break;
    }
}

// ---- JACK callbacks --------------------------------------------------

int AudioEngine::jackProcessCallback(jack_nframes_t nframes, void* arg)
{
    return static_cast<AudioEngine*>(arg)->processAudio(nframes);
}

void AudioEngine::jackShutdownCallback(void* arg)
{
    static_cast<AudioEngine*>(arg)->mRunning.store(false);
    std::cerr << "[AudioEngine] JACK server shut down\n";
}

int AudioEngine::processAudio(jack_nframes_t nframes)
{
    auto* inBuf  = static_cast<float*>(jack_port_get_buffer(mPortIn,   nframes));
    auto* outL   = static_cast<float*>(jack_port_get_buffer(mPortOutL, nframes));
    auto* outR   = static_cast<float*>(jack_port_get_buffer(mPortOutR, nframes));

    const int n = static_cast<int>(nframes);

    // ----------------------------------------------------------------
    // 1. Input gain (smoothed) → mWorkBuf (mono)
    // ----------------------------------------------------------------
    const float targetInGain = dbToLinear(mInputGainDB.load());
    float inPeak = 0.0f;
    for (int i = 0; i < n; ++i)
    {
        mInputGainSmooth = kSmoothCoeff * mInputGainSmooth
                         + (1.0f - kSmoothCoeff) * targetInGain;
        float s = inBuf[i] * mInputGainSmooth;
        mWorkBuf[i] = s;
        if (s < 0.0f) s = -s;
        if (s > inPeak) inPeak = s;
    }
    mInputPeak.store(inPeak);

    // ----------------------------------------------------------------
    // 2. Pre-NAM effects (dynamic order)
    // ----------------------------------------------------------------
    const EffectOrder& order = mOrders[mOrderBuf.load(std::memory_order_acquire)];
    for (int id : order.pre)
        processPreNam(id, mWorkBuf.data(), n);

    // ----------------------------------------------------------------
    // 3. NAM DSP (mono → writes to outL temporarily)
    // ----------------------------------------------------------------
    nam::DSP* model = activeModel();
    if (mModelLoaded.load() && model)
    {
        NAM_SAMPLE* inPtr  = mWorkBuf.data();
        NAM_SAMPLE* outPtr = outL;         // borrow outL as scratch
        NAM_SAMPLE** inPtrs  = &inPtr;
        NAM_SAMPLE** outPtrs = &outPtr;
        model->process(inPtrs, outPtrs, n);
        // Copy NAM output back to mWorkBuf
        std::copy(outL, outL + n, mWorkBuf.begin());
    }

    // ----------------------------------------------------------------
    // 4. 3-band EQ (mono, try_lock)
    // ----------------------------------------------------------------
    if (mEQMutex.try_lock())
    {
        for (int i = 0; i < n; ++i)
            mWorkBuf[i] = mEQ.processSample(mWorkBuf[i]);
        mEQMutex.unlock();
    }

    // ----------------------------------------------------------------
    // 5. Widen to stereo — L = R = mono signal
    // ----------------------------------------------------------------
    std::copy(mWorkBuf.begin(), mWorkBuf.begin() + n, mWorkBufL.begin());
    std::copy(mWorkBuf.begin(), mWorkBuf.begin() + n, mWorkBufR.begin());

    // ----------------------------------------------------------------
    // 6. Post-NAM effects (dynamic order)
    // ----------------------------------------------------------------
    for (int id : order.post)
        processPostNam(id, mWorkBufL.data(), mWorkBufR.data(), n);

    // ----------------------------------------------------------------
    // 7. Output gain (smoothed) + copy to JACK buffers
    // ----------------------------------------------------------------
    const float targetOutGain = dbToLinear(mOutputGainDB.load());
    float outPeak = 0.0f;
    for (int i = 0; i < n; ++i)
    {
        mOutputGainSmooth = kSmoothCoeff * mOutputGainSmooth
                          + (1.0f - kSmoothCoeff) * targetOutGain;
        outL[i] = mWorkBufL[i] * mOutputGainSmooth;
        outR[i] = mWorkBufR[i] * mOutputGainSmooth;
        float peak = std::abs(outL[i]);
        if (peak > outPeak) outPeak = peak;
    }
    mOutputPeak.store(outPeak);

    return 0;
}
// ChorusProcessor.cpp
#include "ChorusProcessor.h"
#include <cmath>

static constexpr float kPi = 3.14159265358979323846f;

void ChorusProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    // Buffer large enough for base delay + max depth + margin
    int bufSamples = static_cast<int>((kBaseMs + kMaxDepthMs + 5.0f) * sr / 1000.0f) + 4;
    mBufL.assign(bufSamples, 0.0f);
    mBufR.assign(bufSamples, 0.0f);
    reset();
}

void ChorusProcessor::reset()
{
    std::fill(mBufL.begin(), mBufL.end(), 0.0f);
    std::fill(mBufR.begin(), mBufR.end(), 0.0f);
    mWriteL = mWriteR = 0;
    mPhase  = 0.0f;
}

float ChorusProcessor::readInterp(const std::vector<float>& buf, int writePos, float delaySamples) const
{
    int sz = static_cast<int>(buf.size());
    float readF = static_cast<float>(writePos) - delaySamples;
    while (readF < 0.0f) readF += sz;

    int   r0   = static_cast<int>(readF) % sz;
    int   r1   = (r0 + 1) % sz;
    float frac = readF - std::floor(readF);
    return buf[r0] * (1.0f - frac) + buf[r1] * frac;
}

void ChorusProcessor::processBlock(float* buf, int nframes)
{
    const float rate  = mRate.load();
    const float depth = mDepth.load() / 100.0f;
    const float mix   = mMix.load()   / 100.0f;
    const float phInc = rate / static_cast<float>(mSampleRate);
    const float fs    = static_cast<float>(mSampleRate);
    const int   sz    = static_cast<int>(mBufL.size());

    for (int i = 0; i < nframes; ++i)
    {
        float lfo     = std::sin(2.0f * kPi * mPhase);
        float delayMs = kBaseMs + depth * kMaxDepthMs * lfo;
        float delaySamples = delayMs * fs / 1000.0f;
        if (delaySamples < 1.0f) delaySamples = 1.0f;

        mBufL[mWriteL] = buf[i];
        float wet = readInterp(mBufL, mWriteL, delaySamples);
        buf[i] = buf[i] * (1.0f - mix) + wet * mix;
        mWriteL = (mWriteL + 1) % sz;

        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}

void ChorusProcessor::processBlockStereo(float* L, float* R, int nframes)
{
    const float rate  = mRate.load();
    const float depth = mDepth.load() / 100.0f;
    const float mix   = mMix.load()   / 100.0f;
    const float phInc = rate / static_cast<float>(mSampleRate);
    const float fs    = static_cast<float>(mSampleRate);
    const int   szL   = static_cast<int>(mBufL.size());
    const int   szR   = static_cast<int>(mBufR.size());

    for (int i = 0; i < nframes; ++i)
    {
        float lfoL = std::sin(2.0f * kPi * mPhase);
        float lfoR = std::sin(2.0f * kPi * (mPhase + 0.5f)); // 180° apart

        float delL = (kBaseMs + depth * kMaxDepthMs * lfoL) * fs / 1000.0f;
        float delR = (kBaseMs + depth * kMaxDepthMs * lfoR) * fs / 1000.0f;
        if (delL < 1.0f) delL = 1.0f;
        if (delR < 1.0f) delR = 1.0f;

        mBufL[mWriteL] = L[i];
        float wetL = readInterp(mBufL, mWriteL, delL);
        L[i] = L[i] * (1.0f - mix) + wetL * mix;
        mWriteL = (mWriteL + 1) % szL;

        mBufR[mWriteR] = R[i];
        float wetR = readInterp(mBufR, mWriteR, delR);
        R[i] = R[i] * (1.0f - mix) + wetR * mix;
        mWriteR = (mWriteR + 1) % szR;

        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}
// CompressorProcessor.cpp
#include "CompressorProcessor.h"
#include <cmath>
#include <algorithm>

void CompressorProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    reset();
}

void CompressorProcessor::reset()
{
    mEnvDB  = -120.0f;
    mGainDB =    0.0f;
}

void CompressorProcessor::processBlock(float* buf, int nframes)
{
    const float threshold = mThresholdDB.load();
    const float ratio     = mRatio.load();
    const float makeup    = mMakeupDB.load();
    const float fs        = static_cast<float>(mSampleRate);

    const float atCoeff  = std::exp(-1.0f / (fs * mAttackMS.load()  * 0.001f));
    const float relCoeff = std::exp(-1.0f / (fs * mReleaseMS.load() * 0.001f));

    for (int i = 0; i < nframes; ++i)
    {
        // Level detection (peak, in dB)
        float level   = std::abs(buf[i]);
        float levelDB = (level > 1e-9f) ? 20.0f * std::log10(level) : -120.0f;

        // Envelope follower in log domain
        if (levelDB > mEnvDB)
            mEnvDB = atCoeff  * mEnvDB + (1.0f - atCoeff)  * levelDB;
        else
            mEnvDB = relCoeff * mEnvDB + (1.0f - relCoeff) * levelDB;

        // Gain computer: compute desired gain reduction
        float targetGR = 0.0f;
        if (mEnvDB > threshold)
            targetGR = (threshold - mEnvDB) * (1.0f - 1.0f / ratio);
        // targetGR <= 0

        // Smooth gain reduction
        if (targetGR < mGainDB)
            mGainDB = atCoeff  * mGainDB + (1.0f - atCoeff)  * targetGR;
        else
            mGainDB = relCoeff * mGainDB + (1.0f - relCoeff) * targetGR;

        float gain = std::pow(10.0f, (mGainDB + makeup) / 20.0f);
        buf[i] *= gain;
    }
}
// DelayProcessor.cpp
#include "DelayProcessor.h"
#include <algorithm>
#include <cmath>

void DelayProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    int maxSamples = static_cast<int>(kMaxDelayMs * sr / 1000.0) + 4;
    mBufL.assign(maxSamples, 0.0f);
    mBufR.assign(maxSamples, 0.0f);
    reset();
}

void DelayProcessor::reset()
{
    std::fill(mBufL.begin(), mBufL.end(), 0.0f);
    std::fill(mBufR.begin(), mBufR.end(), 0.0f);
    mWriteL = mWriteR = 0;
}

void DelayProcessor::processBlock(float* buf, int nframes)
{
    const float timeMS = mTimeMS.load();
    const float fb     = std::min(mFeedback.load() / 100.0f, 0.99f);
    const float mix    = mMix.load() / 100.0f;
    const int   sz     = static_cast<int>(mBufL.size());
    const int   delay  = std::min(static_cast<int>(timeMS * mSampleRate / 1000.0), sz - 1);

    for (int i = 0; i < nframes; ++i)
    {
        int readPos = (mWriteL - delay + sz) % sz;
        float delayed = mBufL[readPos];
        mBufL[mWriteL] = buf[i] + fb * delayed;
        buf[i] = buf[i] * (1.0f - mix) + delayed * mix;
        mWriteL = (mWriteL + 1) % sz;
    }
}

void DelayProcessor::processBlockStereo(float* L, float* R, int nframes)
{
    const float timeMS = mTimeMS.load();
    const float fb     = std::min(mFeedback.load() / 100.0f, 0.99f);
    const float mix    = mMix.load() / 100.0f;
    const int   sz     = static_cast<int>(mBufL.size());
    const int   delay  = std::min(static_cast<int>(timeMS * mSampleRate / 1000.0), sz - 1);

    for (int i = 0; i < nframes; ++i)
    {
        int readPosL = (mWriteL - delay + sz) % sz;
        int readPosR = (mWriteR - delay + sz) % sz;

        float delayedL = mBufL[readPosL];
        float delayedR = mBufR[readPosR];

        // Ping-pong: L feeds into R buffer, R feeds into L buffer
        mBufL[mWriteL] = L[i] + fb * delayedR;
        mBufR[mWriteR] = R[i] + fb * delayedL;

        L[i] = L[i] * (1.0f - mix) + delayedL * mix;
        R[i] = R[i] * (1.0f - mix) + delayedR * mix;

        mWriteL = (mWriteL + 1) % sz;
        mWriteR = (mWriteR + 1) % sz;
    }
}
// EQProcessor.cpp
#include "EQProcessor.h"

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

EQProcessor::EQProcessor()
{
    rebuildFilters();
}

void EQProcessor::setSampleRate(double sampleRate)
{
    mSampleRate = sampleRate;
    rebuildFilters();
    reset();
}

void EQProcessor::setLowGain(double dB)
{
    mLowGainDB = dB;
    mLowShelf = makeLowShelf(mSampleRate, 250.0, dB);
}

void EQProcessor::setMidGain(double dB)
{
    mMidGainDB = dB;
    mMidPeak = makePeaking(mSampleRate, 1000.0, dB, 0.707);
}

void EQProcessor::setTrebleGain(double dB)
{
    mTrebleGainDB = dB;
    mHighShelf = makeHighShelf(mSampleRate, 4000.0, dB);
}

float EQProcessor::processSample(float in)
{
    float s = mLowShelf.process(in);
    s = mMidPeak.process(s);
    s = mHighShelf.process(s);
    return s;
}

void EQProcessor::reset()
{
    mLowShelf.reset();
    mMidPeak.reset();
    mHighShelf.reset();
}

void EQProcessor::rebuildFilters()
{
    mLowShelf  = makeLowShelf(mSampleRate,  250.0,  mLowGainDB);
    mMidPeak   = makePeaking(mSampleRate,  1000.0,  mMidGainDB, 0.707);
    mHighShelf = makeHighShelf(mSampleRate, 4000.0, mTrebleGainDB);
}

// Audio EQ Cookbook biquad designs
EQProcessor::Biquad EQProcessor::makeLowShelf(double sampleRate, double freq, double gainDB)
{
    Biquad bq;
    double A  = std::pow(10.0, gainDB / 40.0);
    double w0 = 2.0 * M_PI * freq / sampleRate;
    double cosW = std::cos(w0);
    double sinW = std::sin(w0);
    double S  = 1.0; // shelf slope
    double alpha = sinW / 2.0 * std::sqrt((A + 1.0 / A) * (1.0 / S - 1.0) + 2.0);

    double b0 =     A * ((A + 1.0) - (A - 1.0) * cosW + 2.0 * std::sqrt(A) * alpha);
    double b1 = 2.0 * A * ((A - 1.0) - (A + 1.0) * cosW);
    double b2 =     A * ((A + 1.0) - (A - 1.0) * cosW - 2.0 * std::sqrt(A) * alpha);
    double a0 =          (A + 1.0) + (A - 1.0) * cosW + 2.0 * std::sqrt(A) * alpha;
    double a1 =    -2.0 * ((A - 1.0) + (A + 1.0) * cosW);
    double a2 =          (A + 1.0) + (A - 1.0) * cosW - 2.0 * std::sqrt(A) * alpha;

    bq.b0 = b0 / a0;
    bq.b1 = b1 / a0;
    bq.b2 = b2 / a0;
    bq.a1 = a1 / a0;
    bq.a2 = a2 / a0;
    return bq;
}

EQProcessor::Biquad EQProcessor::makePeaking(double sampleRate, double freq, double gainDB, double Q)
{
    Biquad bq;
    double A  = std::pow(10.0, gainDB / 40.0);
    double w0 = 2.0 * M_PI * freq / sampleRate;
    double alpha = std::sin(w0) / (2.0 * Q);

    double b0 =  1.0 + alpha * A;
    double b1 = -2.0 * std::cos(w0);
    double b2 =  1.0 - alpha * A;
    double a0 =  1.0 + alpha / A;
    double a1 = -2.0 * std::cos(w0);
    double a2 =  1.0 - alpha / A;

    bq.b0 = b0 / a0;
    bq.b1 = b1 / a0;
    bq.b2 = b2 / a0;
    bq.a1 = a1 / a0;
    bq.a2 = a2 / a0;
    return bq;
}

EQProcessor::Biquad EQProcessor::makeHighShelf(double sampleRate, double freq, double gainDB)
{
    Biquad bq;
    double A  = std::pow(10.0, gainDB / 40.0);
    double w0 = 2.0 * M_PI * freq / sampleRate;
    double cosW = std::cos(w0);
    double sinW = std::sin(w0);
    double S  = 1.0;
    double alpha = sinW / 2.0 * std::sqrt((A + 1.0 / A) * (1.0 / S - 1.0) + 2.0);

    double b0 =      A * ((A + 1.0) + (A - 1.0) * cosW + 2.0 * std::sqrt(A) * alpha);
    double b1 = -2.0 * A * ((A - 1.0) + (A + 1.0) * cosW);
    double b2 =      A * ((A + 1.0) + (A - 1.0) * cosW - 2.0 * std::sqrt(A) * alpha);
    double a0 =           (A + 1.0) - (A - 1.0) * cosW + 2.0 * std::sqrt(A) * alpha;
    double a1 =  2.0 * ((A - 1.0) - (A + 1.0) * cosW);
    double a2 =           (A + 1.0) - (A - 1.0) * cosW - 2.0 * std::sqrt(A) * alpha;

    bq.b0 = b0 / a0;
    bq.b1 = b1 / a0;
    bq.b2 = b2 / a0;
    bq.a1 = a1 / a0;
    bq.a2 = a2 / a0;
    return bq;
}
// FlangerProcessor.cpp
#include "FlangerProcessor.h"
#include <cmath>

static constexpr float kPi = 3.14159265358979323846f;

void FlangerProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    int bufSamples = static_cast<int>((kBaseMs + kDepthMs + 5.0f) * sr / 1000.0f) + 4;
    mBufL.assign(bufSamples, 0.0f);
    mBufR.assign(bufSamples, 0.0f);
    reset();
}

void FlangerProcessor::reset()
{
    std::fill(mBufL.begin(), mBufL.end(), 0.0f);
    std::fill(mBufR.begin(), mBufR.end(), 0.0f);
    mWriteL = mWriteR = 0;
    mFbL    = mFbR    = 0.0f;
    mPhase  = 0.0f;
}

float FlangerProcessor::readInterp(const std::vector<float>& buf, int writePos, float delaySamples) const
{
    int sz = static_cast<int>(buf.size());
    float readF = static_cast<float>(writePos) - delaySamples;
    while (readF < 0.0f) readF += sz;

    int   r0   = static_cast<int>(readF) % sz;
    int   r1   = (r0 + 1) % sz;
    float frac = readF - std::floor(readF);
    return buf[r0] * (1.0f - frac) + buf[r1] * frac;
}

void FlangerProcessor::processSample(float in, float phase,
    std::vector<float>& buf, int& writePos, float& fb,
    float depth, float fbAmt, float mix, float& out) const
{
    float lfo      = std::sin(2.0f * kPi * phase);
    float delayMs  = kBaseMs + depth * kDepthMs * lfo;
    float delaySam = delayMs * static_cast<float>(mSampleRate) / 1000.0f;
    if (delaySam < 1.0f) delaySam = 1.0f;

    int sz       = static_cast<int>(buf.size());
    buf[writePos] = in + fbAmt * fb;
    float wet    = readInterp(buf, writePos, delaySam);
    fb           = wet;
    writePos     = (writePos + 1) % sz;

    out = in * (1.0f - mix) + wet * mix;
}

void FlangerProcessor::processBlock(float* buf, int nframes)
{
    const float depth = mDepth.load()    / 100.0f;
    const float fbAmt = mFeedback.load() / 100.0f;
    const float mix   = mMix.load()      / 100.0f;
    const float phInc = mRate.load()     / static_cast<float>(mSampleRate);

    for (int i = 0; i < nframes; ++i)
    {
        float out;
        processSample(buf[i], mPhase, mBufL, mWriteL, mFbL, depth, fbAmt, mix, out);
        buf[i] = out;

        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}

void FlangerProcessor::processBlockStereo(float* L, float* R, int nframes)
{
    const float depth = mDepth.load()    / 100.0f;
    const float fbAmt = mFeedback.load() / 100.0f;
    const float mix   = mMix.load()      / 100.0f;
    const float phInc = mRate.load()     / static_cast<float>(mSampleRate);

    for (int i = 0; i < nframes; ++i)
    {
        float outL, outR;
        processSample(L[i], mPhase,        mBufL, mWriteL, mFbL, depth, fbAmt, mix, outL);
        processSample(R[i], mPhase + 0.5f, mBufR, mWriteR, mFbR, depth, fbAmt, mix, outR);
        L[i] = outL;
        R[i] = outR;

        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}
// IRLoader.cpp
#include "IRLoader.h"

#include <algorithm>
#include <cstring>

// AudioDSPTools wav loader
#include "wav.h"

IRLoader::IRLoader()
{
    mKernelA.buffer.resize(kMaxIRLength, 0.0f);
    mKernelB.buffer.resize(kMaxIRLength, 0.0f);
}

bool IRLoader::load(const std::string& path)
{
    std::vector<float> rawSamples;
    double sampleRate = 0.0;

    auto result = dsp::wav::Load(path.c_str(), rawSamples, sampleRate);
    if (result != dsp::wav::LoadReturnCode::SUCCESS)
        return false;

    if (rawSamples.empty())
        return false;

    // Take only the first channel if stereo (interleaved), and limit length
    const int irLen = static_cast<int>(std::min<size_t>(rawSamples.size(), kMaxIRLength));

    // Prepare the inactive kernel
    Kernel& k = inactiveKernel();
    k.ir.assign(rawSamples.begin(), rawSamples.begin() + irLen);
    k.irLen   = irLen;
    k.writePos = 0;
    k.buffer.assign(kMaxIRLength, 0.0f);

    // Normalize peak to 1.0
    float peak = 0.0f;
    for (float s : k.ir)
        peak = std::max(peak, std::abs(s));
    if (peak > 1e-6f)
        for (float& s : k.ir)
            s /= peak;

    // Atomic swap
    {
        std::lock_guard<std::mutex> lock(mSwapMutex);
        mActiveIndex.store(mActiveIndex.load() == 0 ? 1 : 0);
    }

    mFilePath = path;
    mLoaded   = true;
    return true;
}

void IRLoader::unload()
{
    mLoaded   = false;
    mFilePath = "";
    mKernelA.irLen = 0;
    mKernelB.irLen = 0;
}

float IRLoader::processSample(float in)
{
    if (!mLoaded)
        return in;

    Kernel& k = activeKernel();
    if (k.irLen <= 0)
        return in;

    // Write input into circular buffer
    k.buffer[k.writePos] = in;

    // Convolve
    float out = 0.0f;
    int   pos = k.writePos;
    for (int i = 0; i < k.irLen; ++i)
    {
        out += k.ir[i] * k.buffer[pos];
        if (--pos < 0)
            pos = k.irLen - 1;
    }

    if (++k.writePos >= k.irLen)
        k.writePos = 0;

    return out;
}

void IRLoader::processBlock(float* samples, int numFrames)
{
    if (!mLoaded)
        return;

    for (int i = 0; i < numFrames; ++i)
        samples[i] = processSample(samples[i]);
}

void IRLoader::reset()
{
    mKernelA.buffer.assign(kMaxIRLength, 0.0f);
    mKernelA.writePos = 0;
    mKernelB.buffer.assign(kMaxIRLength, 0.0f);
    mKernelB.writePos = 0;
}

// ---------------------------------------------------------------------------
// VerticalDial — QDial that maps vertical mouse drag to value changes.
// ---------------------------------------------------------------------------

class VerticalDial : public QDial
{
public:
    explicit VerticalDial(QWidget* parent = nullptr) : QDial(parent)
    {
        setNotchesVisible(true);
        setWrapping(false);
    }

protected:
    void mousePressEvent(QMouseEvent* e) override
    {
        if (e->button() == Qt::LeftButton)
        {
            mDragStartY    = e->pos().y();
            mDragStartVal  = value();
            e->accept();
        }
        else QDial::mousePressEvent(e);
    }

    void mouseMoveEvent(QMouseEvent* e) override
    {
        if (e->buttons() & Qt::LeftButton)
        {
            int dy    = mDragStartY - e->pos().y(); // up → positive
            int range = maximum() - minimum();
            int newVal = mDragStartVal + (dy * range) / 200;
            setValue(qBound(minimum(), newVal, maximum()));
            e->accept();
        }
        else QDial::mouseMoveEvent(e);
    }

    void mouseReleaseEvent(QMouseEvent* e) override
    {
        e->accept(); // do not call base (prevents click-to-jump)
    }

    // Let double-click fall through for the eventFilter in KnobWidget
    void mouseDoubleClickEvent(QMouseEvent* e) override { QDial::mouseDoubleClickEvent(e); }

private:
    int mDragStartY{0};
    int mDragStartVal{0};
};

KnobWidget::KnobWidget(const QString& label, double minVal, double maxVal,
                       double defaultVal, const QString& unit, QWidget* parent)
    : QWidget(parent)
    , mMin(minVal)
    , mMax(maxVal)
    , mDefault(defaultVal)
    , mUnit(unit)
{
    auto* layout = new QVBoxLayout(this);
    layout->setContentsMargins(2, 2, 2, 2);
    layout->setSpacing(1);

    auto* nameLabel = new QLabel(label);
    nameLabel->setAlignment(Qt::AlignHCenter);
    nameLabel->setStyleSheet("font-size: 8pt;");

    mDial = new VerticalDial;
    mDial->setMinimum(0);
    mDial->setMaximum(kSteps);
    mDial->setValue(valueToDial(defaultVal));
    mDial->setSizePolicy(QSizePolicy::Preferred, QSizePolicy::Preferred);

    mLabelValue = new QLabel;
    mLabelValue->setAlignment(Qt::AlignHCenter);
    mLabelValue->setStyleSheet("font-size: 7pt; font-family: monospace;");

    layout->addWidget(nameLabel);
    layout->addWidget(mDial, 1);
    layout->addWidget(mLabelValue);

    updateLabel();
    mDial->installEventFilter(this);
    connect(mDial, &QDial::valueChanged, this, &KnobWidget::onDialChanged);
}

double KnobWidget::value() const { return dialToValue(mDial->value()); }

void KnobWidget::setValue(double v)
{
    v = qBound(mMin, v, mMax);
    const bool blocked = mDial->blockSignals(true);
    mDial->setValue(valueToDial(v));
    mDial->blockSignals(blocked);
    updateLabel();
}

void KnobWidget::resetToDefault()
{
    setValue(mDefault);
    emit valueChanged(mDefault);
}

void KnobWidget::onDialChanged(int dialVal)
{
    updateLabel();
    emit valueChanged(dialToValue(dialVal));
}

void KnobWidget::updateLabel()
{
    double v = value();
    QString txt = mIntDisplay
        ? QString::number(static_cast<int>(std::round(v)))
        : QString::number(v, 'f', 1);
    if (!mUnit.isEmpty())
        txt += " " + mUnit;

    // Use theme highlight color for the value text
    QColor hl = palette().color(QPalette::Highlight);
    mLabelValue->setStyleSheet(
        QString("font-size:7pt; font-family:monospace; color:%1;").arg(hl.name()));
    mLabelValue->setText(txt);
}

bool KnobWidget::eventFilter(QObject* obj, QEvent* event)
{
    if (obj == mDial && event->type() == QEvent::MouseButtonDblClick)
    {
        resetToDefault();
        return true;
    }
    return QWidget::eventFilter(obj, event);
}

double KnobWidget::dialToValue(int d) const
{
    return mMin + (mMax - mMin) * static_cast<double>(d) / kSteps;
}

int KnobWidget::valueToDial(double v) const
{
    double norm = (v - mMin) / (mMax - mMin);
    norm = qBound(0.0, norm, 1.0);
    return static_cast<int>(norm * kSteps + 0.5);
}

// ---------------------------------------------------------------------------
// Effect reorder
// ---------------------------------------------------------------------------

void MainWindow::rebuildEffectOrderFromUI()
{
    EffectOrder order;

    int preIdx  = 0;
    int postIdx = 0;

    for (const auto& er : mEffectRows)
    {
        if (!er.isPost)
        {
            if (preIdx < EffectId::PreNamCount)
                order.pre[preIdx++] = er.effectId;
        }
        else
        {
            if (postIdx < EffectId::PostNamCount)
                order.post[postIdx++] = er.effectId;
        }
    }

    mEngine->setEffectOrder(order);
}

void MainWindow::onMoveUp(int gi)
{
    if (gi <= 0 || gi >= static_cast<int>(mEffectRows.size())) return;

    const bool isPost = mEffectRows[gi].isPost;
    const bool prevIsPost = mEffectRows[gi - 1].isPost;

    // Can't cross the pre/post boundary
    if (isPost != prevIsPost) return;

    std::swap(mEffectRows[gi], mEffectRows[gi - 1]);

    // Move widgets in the layout
    QVBoxLayout* lay = isPost ? mPostChainLayout : mPreChainLayout;
    // Find current positions
    int posA = lay->indexOf(mEffectRows[gi - 1].widget);  // after swap, gi-1 holds what was gi
    int posB = lay->indexOf(mEffectRows[gi].widget);

    if (posA >= 0 && posB >= 0 && posA != posB)
    {
        // Remove the item at the lower index and re-insert above
        auto* itemToMove = lay->takeAt(posB);
        lay->insertItem(posA, itemToMove);
    }

    rebuildEffectOrderFromUI();
}

void MainWindow::onMoveDown(int gi)
{
    if (gi < 0 || gi >= static_cast<int>(mEffectRows.size()) - 1) return;

    const bool isPost = mEffectRows[gi].isPost;
    const bool nextIsPost = mEffectRows[gi + 1].isPost;

    if (isPost != nextIsPost) return;

    std::swap(mEffectRows[gi], mEffectRows[gi + 1]);

    QVBoxLayout* lay = isPost ? mPostChainLayout : mPreChainLayout;
    int posA = lay->indexOf(mEffectRows[gi].widget);      // after swap, gi holds what was gi+1
    int posB = lay->indexOf(mEffectRows[gi + 1].widget);

    if (posA >= 0 && posB >= 0 && posA != posB)
    {
        auto* itemToMove = lay->takeAt(posA);
        lay->insertItem(posB, itemToMove);
    }

    rebuildEffectOrderFromUI();
}

void MainWindow::onLoadNAM()
{
    QString path = QFileDialog::getOpenFileName(
        this, "Load NAM Model", {}, "NAM Files (*.nam);;All Files (*)");
    if (path.isEmpty()) return;
    if (!mEngine->loadModel(path.toStdString()))
    {
        QMessageBox::critical(this, "Error", "Failed to load NAM model:\n" + path);
        return;
    }
    updateStatusLabels();
    mBtnUnloadNAM->setEnabled(true);
}

void MainWindow::onUnloadNAM()
{
    mEngine->unloadModel();
    updateStatusLabels();
    mBtnUnloadNAM->setEnabled(false);
}

void MainWindow::updateMeters()
{
    auto toMeter = [](float peak) -> int {
        if (peak <= 0.0f) return 0;
        float dB = 20.0f * std::log10(peak);
        return std::max(0, std::min(100, static_cast<int>((dB + 60.0f) / 60.0f * 100.0f)));
    };
    mMeterIn ->setValue(toMeter(mEngine->inputPeak()));
    mMeterOut->setValue(toMeter(mEngine->outputPeak()));

    if (!mEngine->isRunning() && mLblJACKStatus->text().startsWith("JACK: connected"))
    {
        mLblJACKStatus->setText("JACK: disconnected");
        mLblJACKStatus->setStyleSheet("color:#f44336; font-size:8pt;");
    }
}

void MainWindow::updateTuner()
{
    if (!mEngine->tunerEnabled()) return;

    float freq = mEngine->tuner().detectPitch();
    if (freq <= 0.0f)
    {
        mLblNote ->setText("--");
        mLblCents->setText("±0¢");
        mTunerBar->setValue(50);
        return;
    }

    int   midi  = TunerProcessor::freqToMidi(freq);
    float cents = TunerProcessor::centsOffset(freq);

    mLblNote ->setText(QString::fromStdString(TunerProcessor::midiToName(midi)));
    mLblCents->setText(QString("%1¢").arg(static_cast<int>(cents)));

    int barVal = static_cast<int>(50.0f + cents);
    mTunerBar->setValue(std::max(0, std::min(100, barVal)));
}

        QMessageBox::critical(this, "Error", "Cannot read preset:\n" + path);

    }

// NoiseGateProcessor.cpp
#include "NoiseGateProcessor.h"
#include <cmath>

void NoiseGateProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    reset();
}

void NoiseGateProcessor::reset()
{
    mEnvelope = 0.0f;
    mGain     = 0.0f;
}

void NoiseGateProcessor::processBlock(float* buf, int nframes)
{
    const float threshLin = std::pow(10.0f, mThresholdDB.load() / 20.0f);
    const float fs        = static_cast<float>(mSampleRate);
    const float atMS      = mAttackMS.load();
    const float relMS     = mReleaseMS.load();

    const float atCoeff  = std::exp(-1.0f / (fs * atMS  * 0.001f));
    const float relCoeff = std::exp(-1.0f / (fs * relMS * 0.001f));

    for (int i = 0; i < nframes; ++i)
    {
        // Peak envelope follower
        float level = std::abs(buf[i]);
        if (level > mEnvelope)
            mEnvelope = atCoeff  * mEnvelope + (1.0f - atCoeff)  * level;
        else
            mEnvelope = relCoeff * mEnvelope + (1.0f - relCoeff) * level;

        // Gate: open when above threshold, close below
        float target = (mEnvelope >= threshLin) ? 1.0f : 0.0f;

        // Smooth gate transitions
        if (target > mGain)
            mGain = atCoeff  * mGain + (1.0f - atCoeff)  * target;
        else
            mGain = relCoeff * mGain + (1.0f - relCoeff) * target;

        buf[i] *= mGain;
    }
}
// PhaserProcessor.cpp
#include "PhaserProcessor.h"
#include <cmath>

static constexpr float kPi = 3.14159265358979323846f;

void PhaserProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    reset();
}

void PhaserProcessor::reset()
{
    mX1L.fill(0.f); mY1L.fill(0.f);
    mX1R.fill(0.f); mY1R.fill(0.f);
    mFbStateL = mFbStateR = 0.0f;
    mPhase    = 0.0f;
}

float PhaserProcessor::processChannel(float in, float phase,
    std::array<float, kStages>& x1,
    std::array<float, kStages>& y1,
    float& fbState,
    float depth, float fbAmt, float mix) const
{
    // LFO drives center frequency on a log scale
    float lfo = 0.5f * (1.0f + std::sin(2.0f * kPi * phase));
    float fc  = kMinFreq * std::pow(kMaxFreq / kMinFreq, depth * lfo);

    // First-order all-pass coefficient
    float w  = std::tan(kPi * fc / static_cast<float>(mSampleRate));
    float d  = (w - 1.0f) / (w + 1.0f);

    // Mix feedback into input
    float sig = in + fbAmt * fbState;

    // Cascade all-pass stages
    for (int s = 0; s < kStages; ++s)
    {
        // y[n] = d*(x[n] - y[n-1]) + x[n-1]
        float y = d * (sig - y1[s]) + x1[s];
        x1[s]  = sig;
        y1[s]  = y;
        sig    = y;
    }

    fbState = sig;
    return in * (1.0f - mix) + sig * mix;
}

void PhaserProcessor::processBlock(float* buf, int nframes)
{
    const float depth = mDepth.load()    / 100.0f;
    const float fbAmt = mFeedback.load() / 100.0f;
    const float mix   = mMix.load()      / 100.0f;
    const float phInc = mRate.load()     / static_cast<float>(mSampleRate);

    for (int i = 0; i < nframes; ++i)
    {
        buf[i] = processChannel(buf[i], mPhase, mX1L, mY1L, mFbStateL, depth, fbAmt, mix);
        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}

void PhaserProcessor::processBlockStereo(float* L, float* R, int nframes)
{
    const float depth = mDepth.load()    / 100.0f;
    const float fbAmt = mFeedback.load() / 100.0f;
    const float mix   = mMix.load()      / 100.0f;
    const float phInc = mRate.load()     / static_cast<float>(mSampleRate);

    for (int i = 0; i < nframes; ++i)
    {
        L[i] = processChannel(L[i], mPhase,        mX1L, mY1L, mFbStateL, depth, fbAmt, mix);
        R[i] = processChannel(R[i], mPhase + 0.5f, mX1R, mY1R, mFbStateR, depth, fbAmt, mix);
        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}
// PitchShiftProcessor.cpp
#include "PitchShiftProcessor.h"
#include <cmath>
#include <algorithm>

static constexpr float kPi = 3.14159265358979323846f;

void PitchShiftProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    mBuf.assign(kBufSize, 0.0f);
    reset();
}

void PitchShiftProcessor::reset()
{
    std::fill(mBuf.begin(), mBuf.end(), 0.0f);
    mWritePos = 0;
    mPhaseA   = 0;
    mPhaseB   = kGrainSize / 2;
    mReadPosA = static_cast<float>((kBufSize - kGrainSize) % kBufSize);
    mReadPosB = static_cast<float>((kBufSize - kGrainSize + kGrainSize / 2) % kBufSize);
}

float PitchShiftProcessor::readInterp(float pos) const
{
    // Wrap position into [0, kBufSize)
    while (pos < 0.f)      pos += kBufSize;
    while (pos >= kBufSize) pos -= kBufSize;

    int   r0   = static_cast<int>(pos) % kBufSize;
    int   r1   = (r0 + 1) % kBufSize;
    float frac = pos - std::floor(pos);
    return mBuf[r0] * (1.0f - frac) + mBuf[r1] * frac;
}

void PitchShiftProcessor::processBlock(float* buf, int nframes)
{
    const float semitones = std::round(mSemitones.load());
    const float mix       = mMix.load() / 100.0f;

    if (std::abs(mix) < 1e-4f && std::abs(semitones) < 0.5f)
        return; // fully dry and no shift

    const float ratio = std::pow(2.0f, semitones / 12.0f);

    for (int i = 0; i < nframes; ++i)
    {
        const float dry = buf[i];

        // Write input into circular buffer
        mBuf[mWritePos] = dry;

        // --- Grain A ---
        // Hann window: w = 0.5*(1 - cos(2*pi*phase/G))
        float winA = 0.5f * (1.0f - std::cos(2.0f * kPi * mPhaseA / kGrainSize));
        float smpA = readInterp(mReadPosA) * winA;

        // --- Grain B (180° offset) ---
        float winB = 0.5f * (1.0f - std::cos(2.0f * kPi * mPhaseB / kGrainSize));
        float smpB = readInterp(mReadPosB) * winB;

        float wet = smpA + smpB;

        buf[i] = dry * (1.0f - mix) + wet * mix;

        // Advance write pointer
        mWritePos = (mWritePos + 1) % kBufSize;

        // Advance read pointers at pitch ratio speed
        mReadPosA += ratio;
        if (mReadPosA >= kBufSize) mReadPosA -= kBufSize;

        mReadPosB += ratio;
        if (mReadPosB >= kBufSize) mReadPosB -= kBufSize;

        // Advance grain phases
        ++mPhaseA;
        ++mPhaseB;

        // Reset grain A when it completes: jump read pointer back to
        // kGrainSize samples behind write pointer (re-syncs with input)
        if (mPhaseA >= kGrainSize)
        {
            mPhaseA   = 0;
            mReadPosA = static_cast<float>((mWritePos - kGrainSize + kBufSize) % kBufSize);
        }

        // Reset grain B
        if (mPhaseB >= kGrainSize)
        {
            mPhaseB   = 0;
            mReadPosB = static_cast<float>((mWritePos - kGrainSize + kBufSize) % kBufSize);
        }
    }
}
// ReverbProcessor.cpp
#include "ReverbProcessor.h"
#include <cmath>

constexpr int ReverbProcessor::kCombLengths[kNumCombs];
constexpr int ReverbProcessor::kApLengths[kNumAllPasses];

ReverbProcessor::ReverbProcessor()
{
    allocate();
}

void ReverbProcessor::allocate()
{
    double sr = mSampleRate;
    double scale = sr / 44100.0;
    for (int i = 0; i < kNumCombs; ++i)
    {
        mCombL[i].resize(static_cast<int>(kCombLengths[i] * scale));
        mCombR[i].resize(static_cast<int>((kCombLengths[i] + kStereoOffset) * scale));
    }
    for (int i = 0; i < kNumAllPasses; ++i)
    {
        mApL[i].resize(static_cast<int>(kApLengths[i] * scale));
        mApR[i].resize(static_cast<int>((kApLengths[i] + kStereoOffset) * scale));
    }
}

void ReverbProcessor::setSampleRate(double sr)
{
    mSampleRate = sr;
    allocate();
    reset();
}

void ReverbProcessor::reset()
{
    for (auto& c : mCombL) c.reset();
    for (auto& c : mCombR) c.reset();
    for (auto& a : mApL)   a.reset();
    for (auto& a : mApR)   a.reset();
}

void ReverbProcessor::processBlockStereo(float* L, float* R, int nframes)
{
    // Map parameters
    const float roomSize = mRoomSize.load() / 100.0f;
    const float damping  = mDamping.load()  / 100.0f;
    const float width    = mWidth.load()    / 100.0f;
    const float mix      = mMix.load()      / 100.0f;

    // Freeverb feedback/damping coefficients
    const float feedback = 0.28f + roomSize * 0.7f;
    const float damp     = damping * 0.4f;

    for (int i = 0; i < nframes; ++i)
    {
        float inMono = (L[i] + R[i]) * 0.015f; // attenuate to prevent clip

        // Run comb filters in parallel
        float outL = 0.0f, outR = 0.0f;
        for (int c = 0; c < kNumCombs; ++c)
        {
            outL += mCombL[c].process(inMono, feedback, damp);
            outR += mCombR[c].process(inMono, feedback, damp);
        }

        // Run all-pass filters in series
        for (int a = 0; a < kNumAllPasses; ++a)
        {
            outL = mApL[a].process(outL);
            outR = mApR[a].process(outR);
        }

        // Width: blend L and R wet signals
        float wet1 = width * 0.5f + 0.5f;
        float wet2 = (1.0f - width) * 0.5f;
        float wetL = outL * wet1 + outR * wet2;
        float wetR = outR * wet1 + outL * wet2;

        L[i] = L[i] * (1.0f - mix) + wetL * mix;
        R[i] = R[i] * (1.0f - mix) + wetR * mix;
    }
}

void ReverbProcessor::processBlock(float* buf, int nframes)
{
    // Mono: run L path only
    const float roomSize = mRoomSize.load() / 100.0f;
    const float damping  = mDamping.load()  / 100.0f;
    const float mix      = mMix.load()      / 100.0f;
    const float feedback = 0.28f + roomSize * 0.7f;
    const float damp     = damping * 0.4f;

    for (int i = 0; i < nframes; ++i)
    {
        float inSig = buf[i] * 0.015f;
        float out   = 0.0f;
        for (int c = 0; c < kNumCombs; ++c)
            out += mCombL[c].process(inSig, feedback, damp);
        for (int a = 0; a < kNumAllPasses; ++a)
            out = mApL[a].process(out);
        buf[i] = buf[i] * (1.0f - mix) + out * mix;
    }
}

#include "TremoloProcessor.h"
#include <cmath>

static constexpr float kPi = 3.14159265358979323846f;

void TremoloProcessor::processBlock(float* buf, int nframes)
{
    const float rate  = mRate.load();
    const float depth = mDepth.load() / 100.0f;
    const float phInc = rate / static_cast<float>(mSampleRate);

    for (int i = 0; i < nframes; ++i)
    {
        // LFO goes -1 to +1; scale to gain 1-depth to 1
        float lfo  = std::sin(2.0f * kPi * mPhase);
        float gain = 1.0f - depth * (1.0f - lfo) * 0.5f;
        buf[i] *= gain;

        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}

void TremoloProcessor::processBlockStereo(float* L, float* R, int nframes)
{
    const float rate  = mRate.load();
    const float depth = mDepth.load() / 100.0f;
    const float phInc = rate / static_cast<float>(mSampleRate);

    for (int i = 0; i < nframes; ++i)
    {
        // L and R: 90° apart for gentle auto-pan effect
        float lfoL  = std::sin(2.0f * kPi * mPhase);
        float lfoR  = std::sin(2.0f * kPi * (mPhase + 0.25f));
        float gainL = 1.0f - depth * (1.0f - lfoL) * 0.5f;
        float gainR = 1.0f - depth * (1.0f - lfoR) * 0.5f;
        L[i] *= gainL;
        R[i] *= gainR;

        mPhase += phInc;
        if (mPhase >= 1.0f) mPhase -= 1.0f;
    }
}
// TunerProcessor.cpp
#include "TunerProcessor.h"

#include <cmath>
#include <algorithm>

TunerProcessor::TunerProcessor()
{
    mRingBuf.fill(0.f);
    mSnapBuf.fill(0.f);
}

void TunerProcessor::reset()
{
    mRingBuf.fill(0.f);
    mWritePos.store(0);
    mBufferReady.store(false);
}

void TunerProcessor::processBlock(float* buf, int nframes, bool mute)
{
    int wp = mWritePos.load(std::memory_order_relaxed);
    for (int i = 0; i < nframes; ++i)
    {
        mRingBuf[wp] = buf[i];
        wp = (wp + 1) % kBufSize;
        if (mute) buf[i] = 0.0f;
    }
    mWritePos.store(wp, std::memory_order_release);
    mBufferReady.store(true, std::memory_order_release);
}

float TunerProcessor::detectPitch()
{
    if (!mBufferReady.load(std::memory_order_acquire))
        return 0.0f;

    // Copy ring buffer into contiguous snapshot (oldest → newest)
    const int startPos = mWritePos.load(std::memory_order_acquire);
    for (int i = 0; i < kBufSize; ++i)
        mSnapBuf[i] = mRingBuf[(startPos + i) % kBufSize];

    // RMS check — silence guard
    double rmsSum = 0.0;
    for (int i = 0; i < kBufSize; ++i)
        rmsSum += mSnapBuf[i] * mSnapBuf[i];
    const double rms = std::sqrt(rmsSum / kBufSize);
    if (rms < 5e-4)
        return 0.0f;

    // Guitar range: E2 (82 Hz) to E6 (1319 Hz)
    const int fs      = static_cast<int>(mSampleRate);
    const int minLag  = fs / 1400;       // ~31 samples at 44100
    const int maxLag  = fs / 50;         // ~882 samples at 44100
    const int N       = kBufSize;

    // --- Normalized Square Difference Function (NSDF / YIN-style) ---
    // nsdf[k] = 2*r[k] / (m0 + mk)
    // where r[k] = correlation, m0 = energy of first half, mk = energy of second half.
    // nsdf ranges from -1 to +1; 1 = perfect periodicity.

    // Pre-compute running energy sums using a sliding window
    // m0 = sum x[n]^2 for n in [0, N-lag)
    // mk = sum x[n+lag]^2 for n in [0, N-lag)

    float bestNSDF = -1.0f;
    int   bestLag  = 0;

    // Use a reasonable window size to avoid O(N^2) being too slow
    const int winLen = std::min(N, 1024);

    double r0 = 0.0;
    for (int i = 0; i < winLen; ++i)
        r0 += mSnapBuf[i] * mSnapBuf[i];

    for (int lag = minLag; lag <= std::min(maxLag, N - winLen - 1); ++lag)
    {
        double r  = 0.0;
        double mk = 0.0;
        for (int i = 0; i < winLen; ++i)
        {
            r  += mSnapBuf[i] * mSnapBuf[i + lag];
            mk += mSnapBuf[i + lag] * mSnapBuf[i + lag];
        }

        double denom = r0 + mk;
        float  nsdf  = (denom > 1e-12) ? static_cast<float>(2.0 * r / denom) : 0.0f;

        if (nsdf > bestNSDF)
        {
            bestNSDF = nsdf;
            bestLag  = lag;
        }
    }

    // Require at least 40% periodicity
    if (bestNSDF < 0.40f || bestLag == 0)
        return 0.0f;

    // Parabolic interpolation around bestLag for sub-sample accuracy
    if (bestLag > minLag && bestLag < std::min(maxLag, N - winLen - 2))
    {
        auto nsdfAt = [&](int lag) -> float {
            double r = 0.0, mk = 0.0;
            for (int i = 0; i < winLen; ++i)
            {
                r  += mSnapBuf[i] * mSnapBuf[i + lag];
                mk += mSnapBuf[i + lag] * mSnapBuf[i + lag];
            }
            double d = r0 + mk;
            return (d > 1e-12) ? static_cast<float>(2.0 * r / d) : 0.0f;
        };

        float y0 = nsdfAt(bestLag - 1);
        float y1 = bestNSDF;
        float y2 = nsdfAt(bestLag + 1);

        float denom = 2.0f * (2.0f * y1 - y0 - y2);
        if (std::abs(denom) > 1e-6f)
        {
            float delta = (y0 - y2) / denom;
            // clamp interpolation correction to ±1 sample
            delta = std::max(-1.0f, std::min(1.0f, delta));
            return static_cast<float>(mSampleRate) / (bestLag + delta);
        }
    }

    return static_cast<float>(mSampleRate) / bestLag;
}

int TunerProcessor::freqToMidi(float freq)
{
    if (freq <= 0.0f) return -1;
    return static_cast<int>(std::round(69.0f + 12.0f * std::log2(freq / 440.0f)));
}

std::string TunerProcessor::midiToName(int midiNote)
{
    if (midiNote < 0 || midiNote > 127) return "--";
    static const char* names[] = {"C","C#","D","D#","E","F","F#","G","G#","A","A#","B"};
    int octave = midiNote / 12 - 1;
    return std::string(names[midiNote % 12]) + std::to_string(octave);
}

float TunerProcessor::centsOffset(float freq)
{
    if (freq <= 0.0f) return 0.0f;
    int   midi     = freqToMidi(freq);
    float noteFreq = 440.0f * std::pow(2.0f, (midi - 69) / 12.0f);
    float cents    = 1200.0f * std::log2(freq / noteFreq);
    return std::max(-50.0f, std::min(50.0f, cents));
}

/// Abstract base for all audio effect processors.
/// All effects support both mono and stereo block processing.
/// Default stereo implementation calls processBlock on each channel independently.
class EffectBase
{
public:
    virtual ~EffectBase() = default;

    virtual void setSampleRate(double sr) = 0;
    virtual void reset() = 0;

/// Simple 3-band parametric EQ implemented as biquad filters.
/// Bands: Low shelf, Mid peak, High shelf
class EQProcessor
{
public:
    EQProcessor();

    void setSampleRate(double sampleRate);

    /// Gain in dB, range typically -12 to +12
    void setLowGain(double dB);
    void setMidGain(double dB);
    void setTrebleGain(double dB);

    /// Process a single mono sample in-place
    float processSample(float in);

    /// Reset filter state
    void reset();

private:
    struct Biquad
    {
        // Coefficients
        double b0 = 1, b1 = 0, b2 = 0, a1 = 0, a2 = 0;
        // State
        double x1 = 0, x2 = 0, y1 = 0, y2 = 0;

        float process(float in)
        {
            double x0 = in;
            double y0 = b0 * x0 + b1 * x1 + b2 * x2 - a1 * y1 - a2 * y2;
            x2 = x1; x1 = x0;
            y2 = y1; y1 = y0;
            return static_cast<float>(y0);
        }

        void reset() { x1 = x2 = y1 = y2 = 0.0; }
    };

    /// Design a low-shelf biquad at the given frequency
    static Biquad makeLowShelf(double sampleRate, double freq, double gainDB);
    /// Design a peaking EQ biquad at the given frequency
    static Biquad makePeaking(double sampleRate, double freq, double gainDB, double Q);
    /// Design a high-shelf biquad at the given frequency
    static Biquad makeHighShelf(double sampleRate, double freq, double gainDB);

    double mSampleRate = 44100.0;
    double mLowGainDB  = 0.0;
    double mMidGainDB  = 0.0;
    double mTrebleGainDB = 0.0;

    Biquad mLowShelf;
    Biquad mMidPeak;
    Biquad mHighShelf;

    void rebuildFilters();
};

/// Loads an IR (Impulse Response) WAV file and applies it as a convolution
/// cab simulator using a simple brute-force FIR convolution.
/// The IR is truncated to a reasonable length (e.g. 4096 samples) for performance.
class IRLoader
{
public:
    static constexpr int kMaxIRLength = 4096;

    IRLoader();

    /// Load an IR from a WAV file. Thread-safe: swaps the kernel atomically.
    /// Returns true on success.
    bool load(const std::string& path);

    /// Unload the current IR (bypass mode)
    void unload();

    bool isLoaded() const { return mLoaded; }
    std::string getFilePath() const { return mFilePath; }

    /// Process a single sample (call in audio thread, not real-time safe if load() is
    /// called concurrently -- use double-buffered swap below).
    float processSample(float in);

    /// Process a block of samples in-place. Uses the active kernel.
    void processBlock(float* samples, int numFrames);

    /// Reset convolution state
    void reset();

private:
    bool mLoaded = false;
    std::string mFilePath;

    // Double-buffered kernels: audio thread reads from mActive, GUI thread writes mPending
    struct Kernel
    {
        std::vector<float> ir;
        std::vector<float> buffer; // circular delay line

        int writePos = 0;
        int irLen    = 0;
    };

    Kernel mKernelA;
    Kernel mKernelB;

    // Which kernel is active in the audio thread (0=A, 1=B)
    std::atomic<int> mActiveIndex{0};
    std::mutex       mSwapMutex;

    Kernel& activeKernel()   { return mActiveIndex.load() == 0 ? mKernelA : mKernelB; }
    Kernel& inactiveKernel() { return mActiveIndex.load() == 0 ? mKernelB : mKernelA; }
};

/// Stereo reverb based on the Freeverb algorithm.
/// (8 parallel comb filters → 4 series all-pass filters, stereo.)
class ReverbProcessor : public EffectBase
{
public:
    ReverbProcessor();

    void setSampleRate(double sr) override;
    void reset() override;

    // Stereo is the natural mode; mono falls back to L-only
    void processBlock(float* buf, int nframes) override;
    void processBlockStereo(float* L, float* R, int nframes) override;

    void setRoomSize(float pct)  { mRoomSize.store(pct);  }  // 0–100
    void setDamping(float pct)   { mDamping.store(pct);   }  // 0–100
    void setWidth(float pct)     { mWidth.store(pct);     }  // 0–100
    void setMix(float pct)       { mMix.store(pct);       }  // 0–100

    float roomSize() const { return mRoomSize.load(); }
    float damping()  const { return mDamping.load();  }
    float width()    const { return mWidth.load();    }
    float mix()      const { return mMix.load();      }

private:
    std::atomic<float> mRoomSize{50.0f};
    std::atomic<float> mDamping {50.0f};
    std::atomic<float> mWidth   {100.0f};
    std::atomic<float> mMix     {30.0f};

    double mSampleRate{44100.0};

    // Freeverb comb filter
    struct CombFilter
    {
        std::vector<float> buf;
        int    writePos{0};
        float  filterStore{0.0f};

        void resize(int n) { buf.assign(n, 0.0f); }
        void reset() { std::fill(buf.begin(), buf.end(), 0.0f); filterStore = 0.0f; writePos = 0; }

        float process(float in, float feedback, float damp)
        {
            int sz     = static_cast<int>(buf.size());
            float out  = buf[writePos];
            filterStore = out * (1.0f - damp) + filterStore * damp;
            buf[writePos] = in + filterStore * feedback;
            writePos = (writePos + 1) % sz;
            return out;
        }
    };

    // Freeverb all-pass filter
    struct AllPassFilter
    {
        std::vector<float> buf;
        int writePos{0};
        static constexpr float kFeedback = 0.5f;

        void resize(int n) { buf.assign(n, 0.0f); }
        void reset() { std::fill(buf.begin(), buf.end(), 0.0f); writePos = 0; }

        float process(float in)
        {
            int   sz      = static_cast<int>(buf.size());
            float bufOut  = buf[writePos];
            float output  = -in + bufOut;
            buf[writePos] = in + bufOut * kFeedback;
            writePos      = (writePos + 1) % sz;
            return output;
        }
    };

    static constexpr int kNumCombs    = 8;
    static constexpr int kNumAllPasses = 4;

    // Stereo: separate L/R comb + allpass banks
    std::array<CombFilter,    kNumCombs>     mCombL, mCombR;
    std::array<AllPassFilter, kNumAllPasses> mApL,   mApR;

    // Freeverb delay lengths at 44100 Hz
    static constexpr int kCombLengths[kNumCombs]    = {1116,1188,1277,1356,1422,1491,1557,1617};
    static constexpr int kApLengths[kNumAllPasses]  = {556, 441, 341, 225};
    static constexpr int kStereoOffset              = 23; // samples offset for R channel

    void allocate();
};

    static constexpr int kBufSize = 4096;

    TunerProcessor();

    void setSampleRate(double sr) { mSampleRate = sr; }
    void reset();

    // Called from audio thread — accumulates samples, optionally mutes
    void processBlock(float* buf, int nframes, bool mute);

    // Called from GUI thread — runs pitch detection, returns detected frequency
    // Returns 0.0 if not enough data or no pitch detected
    float detectPitch();

    // Converts frequency to MIDI note number (A4=69=440Hz)
    static int  freqToMidi(float freq);
    // Returns note name string, e.g. "A4", "C#3"
    static std::string midiToName(int midiNote);
    // Returns cents offset from nearest semitone (-50 to +50)
    static float centsOffset(float freq);

private:
    double mSampleRate{44100.0};

    // Ring buffer filled by audio thread, consumed by GUI thread
    std::array<float, kBufSize> mRingBuf{};
    std::atomic<int>            mWritePos{0};
    std::atomic<bool>           mBufferReady{false};

    // Snapshot used by detectPitch()
    std::array<float, kBufSize> mSnapBuf{};
};
